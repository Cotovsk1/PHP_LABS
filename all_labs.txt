/**
 * Форматує запис рецепта у вигляді короткого інформативного рядка.
 *
 * @param array $recipe Асоціативний масив рецепта
 * @return string Відформатований рядок опису
 */
function formatRecipe(array $recipe): string {
    return "{$recipe['title']} — {$recipe['servings']} порц. ({$recipe['cookTimeMin']} хв, {$recipe['difficulty']})";
}


/**
 * Визначає мітку тривалості приготування рецепта на основі умови.
 *
 * @param int $cookTimeMin Час приготування у хвилинах
 * @return array Масив із текстовою міткою, css-класом та ознакою швидкості
 */
function getRecipeBadge(int $cookTimeMin): array {
    if ($cookTimeMin <= 20) {
        return [
            'label' => '⚡ Швидкий рецепт',
            'class' => 'badge-quick',
            'isQuick' => true,
        ];
    } elseif ($cookTimeMin <= 45) {
        return [
            'label' => '⏱ Помірний час',
            'class' => 'badge-medium',
            'isQuick' => false,
        ];
    } else {
        return [
            'label' => '⏳ Тривале готування',
            'class' => 'badge-slow',
            'isQuick' => false,
        ];
    }
}

# Практична робота: Знайомство з базовим синтаксисом PHP

## Варіант 6. Кулінарна книга (рецепти, інгредієнти)

### Вимоги варіанта:
- **Дані**: масив рецептів — `title`, `cookTimeMin`, `servings`, `difficulty` (+ `ingredients`, `category`).
- **Умова**: якщо `cookTimeMin <= 20` — мітка «Швидкий рецепт».
- **Агрегат**: середній час приготування по всіх рецептах.

---

### Структура проєкту:
- [`index.php`](./index.php) — головний PHP-скрипт з масивом даних, типізованою функцією, умовною логікою, розрахунком агрегатів та HTML-шаблоном.
- [`style.css`](./style.css) — файл стилів із адаптивною сіткою карток, кольоровими бейджами для міток та таблицею.
- [`run_server.bat`](./run_server.bat) — скрипт для швидкого запуску вбудованого вебсервера PHP (`php -S localhost:8000`).

---

    /**
     * Оновлює візуальні індикатори та текст поточної одиниці виміру
     */
    function updateUnitDisplay(unitValue) {
        const readable = unitLabels[unitValue] || unitValue;
        if (activeUnitIndicator) {
            activeUnitIndicator.innerHTML = `Поточна міра: <strong>${readable}</strong>`;
        }
        if (storagePreview) {
            const saved = localStorage.getItem(STORAGE_KEY_UNIT);
            if (saved) {
                storagePreview.textContent = `${STORAGE_KEY_UNIT} = "${saved}" (${unitLabels[saved] || saved})`;
                storagePreview.classList.add('has-value');
            } else {
                storagePreview.textContent = 'немає запису (за замовчуванням: грами)';
                storagePreview.classList.remove('has-value');
            }
        }
    }

     * Відновлення збереженої одиниці виміру з localStorage
     */
    function restoreSavedUnit() {
        const savedUnit = localStorage.getItem(STORAGE_KEY_UNIT);
        if (savedUnit && (savedUnit === 'g' || savedUnit === 'oz')) {
            unitRadios.forEach(radio => {
                if (radio.value === savedUnit) {
                    radio.checked = true;

                    $result[] = $recipe;
                    break;
                }
            }
        }
        return $result;
    }

    public function shortestCookTime(): ?Recipe
    {

        foreach ($this->recipes as $recipe) {
            if ($shortest === null || $recipe->getCookTimeMin() < $shortest->getCookTimeMin()) {

            }
        }

    public function __construct(
        string $title,
        array $ingredients,
        int $cookTimeMin,
        int $servings = 2,
        array $substitutions = []
    ) {
        parent::__construct($title, $ingredients, $cookTimeMin, $servings);
        $this->substitutions = $substitutions;
    }

    public function getSubstitutions(): array
    {
        return $this->substitutions;
    }

    {
        return 'Веганський';
    }

    public function getInfo(): string
    {
        $info = parent::getInfo() . ' | 🌱 веганський';

        if ($this->substitutions === []) {
            return $info . ', без замін';
        }

        $pairs = [];
        foreach ($this->substitutions as $from => $to) {
            $pairs[] = "{$from} → {$to}";
        }

        return $info . '; заміни: ' . implode(', ', $pairs);
    }

if (!function_exists('convertGramsToOz')) {
    /** Переводить грами в унції (1 oz = 28,3495 г), округлення до 0,1. */
    function convertGramsToOz(float $grams): float
    {
        return round($grams / 28.3495, 1);
    }
}

if (!function_exists('formatIngredientsList')) {
    /**
     * Форматує інгредієнти («назва => грами») у рядок: «Борошно — 250 г, Молоко — 300 г».
     * $unit: 'g' (грами) або 'oz' (унції).
     */
    function formatIngredientsList(array $ingredients, string $unit = 'g'): string
    {
        if ($ingredients === []) {
            return '—';
        }

        $parts = [];
        foreach ($ingredients as $name => $grams) {
            $amount  = $unit === 'oz'
                ? convertGramsToOz((float) $grams) . ' унц.'
                : $grams . ' г';
            $parts[] = "{$name} — {$amount}";
        }
        return implode(', ', $parts);
    }
}

if (!function_exists('formatCookTime')) {
    /** 90 → «1 год 30 хв», 45 → «45 хв», 120 → «2 год». */
    function formatCookTime(int $minutes): string
    {
        $h = intdiv($minutes, 60);
        $m = $minutes % 60;

        if ($h === 0) {
            return "{$minutes} хв";
        }
        return $m === 0 ? "{$h} год" : "{$h} год {$m} хв";
    }
}

declare(strict_types=1);

/**
 * Практична робота №4: Функції для роботи з базою даних рецептів (Варіант 6)
 * Усі операції виконуються виключно через підготовлені запити (prepared statements).
 */

/**
 * Отримати всі рецепти (SELECT * FROM recipes)
 * 
 * @param PDO $pdo
 * @return array
 */
function getAllRecipes(PDO $pdo): array
{
    $stmt = $pdo->query('SELECT * FROM recipes ORDER BY id DESC');
    return $stmt->fetchAll();
}

/**
 * Отримати один рецепт за його первинним ключем (id)
 * 
 * @param PDO $pdo
 * @param int $id
 * @return array|null
 */
function getRecipeById(PDO $pdo, int $id): ?array
{
    $stmt = $pdo->prepare('SELECT * FROM recipes WHERE id = :id');
    $stmt->execute([':id' => $id]);
    $recipe = $stmt->fetch();
    return $recipe ?: null;
}

/**
 * Вибірка за інгредієнтом (Вимога варіанта 6):
 * findByIngredient($text) - SELECT * FROM recipes WHERE ingredients LIKE :pattern
 * 
 * @param PDO $pdo
 * @param string $text
 * @return array
 */
function findByIngredient(PDO $pdo, string $text): array
{
    $stmt = $pdo->prepare('SELECT * FROM recipes WHERE ingredients LIKE :pattern ORDER BY id DESC');
    // Використання плейсхолдера унеможливлює SQL-ін'єкцію через рядок пошуку
    $stmt->execute([':pattern' => '%' . $text . '%']);
    return $stmt->fetchAll();
}

/**
 * Вибірка найшвидшого рецепта (Вимога варіанта 6):
 * shortestCookTime() - SELECT * FROM recipes ORDER BY cook_time_min ASC LIMIT 1
 * 
 * @param PDO $pdo
 * @return array|null
 */
function shortestCookTime(PDO $pdo): ?array
{
    $stmt = $pdo->query('SELECT * FROM recipes ORDER BY cook_time_min ASC LIMIT 1');
    $recipe = $stmt->fetch();
    return $recipe ?: null;
}

/**
 * Додавання рецепта (Вимога варіанта 6):
 * addRecipe() - INSERT
 * 
 * @param PDO $pdo
 * @param string $title
 * @param string $ingredients
 * @param int $cookTimeMin
 * @return int ID новоствореного рядка
 */
function addRecipe(PDO $pdo, string $title, string $ingredients, int $cookTimeMin): int
{
    $sql = 'INSERT INTO recipes (title, ingredients, cook_time_min) VALUES (:title, :ingredients, :cook_time_min)';
    $stmt = $pdo->prepare($sql);
    $stmt->execute([
        ':title'         => $title,
        ':ingredients'   => $ingredients,
        ':cook_time_min' => $cookTimeMin,
    ]);
    return (int) $pdo->lastInsertId();
}

/**
 * Оновлення рецепта (Вимога варіанта 6):
 * updateRecipe($id, ...) - UPDATE ... WHERE id = :id
 * 
 * @param PDO $pdo
 * @param int $id
 * @param string $title
 * @param string $ingredients
 * @param int $cookTimeMin
 * @return bool
 */
function updateRecipe(PDO $pdo, int $id, string $title, string $ingredients, int $cookTimeMin): bool
{
    $sql = 'UPDATE recipes SET title = :title, ingredients = :ingredients, cook_time_min = :cook_time_min WHERE id = :id';
    $stmt = $pdo->prepare($sql);
    return $stmt->execute([
        ':id'            => $id,
        ':title'         => $title,
        ':ingredients'   => $ingredients,
        ':cook_time_min' => $cookTimeMin,
    ]);
}

/**
 * Видалення рецепта (Вимога варіанта 6):
 * deleteRecipe($id) - DELETE FROM recipes WHERE id = :id
 * 
 * @param PDO $pdo
 * @param int $id
 * @return bool
 */
function deleteRecipe(PDO $pdo, int $id): bool
{
    $stmt = $pdo->prepare('DELETE FROM recipes WHERE id = :id');
    return $stmt->execute([':id' => $id]);
}

/**

# Практична робота №4: Взаємодія вебзастосунку з базою даних (PDO/MySQL)

## 🎯 Мета роботи
- Опанувати взаємодію вебзастосунку з СУБД MySQL/MariaDB засобами розширення **PDO (PHP Data Objects)**.
- Створити базу даних та таблицю `recipes` відповідно до вимог варіанта №6.
- Реалізувати повний цикл базових CRUD-операцій:
  - **C**reate (`INSERT`)
  - **R**ead (`SELECT`)
  - **U**pdate (`UPDATE`)
  - **D**elete (`DELETE`)
- Застосувати підготовлені запити (**prepared statements**) з іменованими плейсхолдерами для надійного захисту від SQL-ін'єкцій.
- Реалізувати спеціальні предметні вибірки варіанта:
  - `findByIngredient($text)` — пошук рецептів за інгредієнтом із шаблоном `LIKE :pattern`.
  - `shortestCookTime()` — визначення рецепта з мінімальним часом приготування.

### Крок 1. Створення бази даних та таблиці (`schema.sql`)
Створено базу даних `practicum4` та таблицю `recipes` згідно зі специфікацією варіанта №6:

```sql
CREATE DATABASE IF NOT EXISTS `practicum4`

CREATE TABLE IF NOT EXISTS `recipes` (
```

```php
$dsn = "mysql:host={$host};port={$port};dbname={$dbname};charset=utf8mb4";
$options = [
    PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,

```

```php
$stmt = $pdo->query('SELECT * FROM recipes ORDER BY id DESC');

```
Додатково реалізовано вибірки варіанта №6:
* **Пошук за інгредієнтом:**
  ```php
  function findByIngredient(PDO $pdo, string $text): array {
      $stmt = $pdo->prepare('SELECT * FROM recipes WHERE ingredients LIKE :pattern ORDER BY id DESC');
      $stmt->execute([':pattern' => '%' . $text . '%']);
      return $stmt->fetchAll();
  }
  ```
* **Найшвидший рецепт:**
  ```php
  function shortestCookTime(PDO $pdo): ?array {
      $stmt = $pdo->query('SELECT * FROM recipes ORDER BY cook_time_min ASC LIMIT 1');
      return $stmt->fetch() ?: null;
  }
  ```

### Крок 4. Додавання рецепта (INSERT)
У файлі `add.php` розміщено HTML-форму. Після відправлення виконується валідація та вставка через підготовлений запит:
```php
$stmt = $pdo->prepare('INSERT INTO recipes (title, ingredients, cook_time_min) VALUES (:title, :ingredients, :cook_time_min)');
$stmt->execute([
    ':title'         => $title,
    ':ingredients'   => $ingredients,
    ':cook_time_min' => $cookTimeMin,
]);
header('Location: index.php?msg=added');
exit;
```

### Крок 5. Редагування рецепта (UPDATE)
У файлі `edit.php` завантажуються поточні дані (`SELECT ... WHERE id = :id`), заповнюються у поля форми, а після відправлення оновлюються:
```php
$stmt = $pdo->prepare('UPDATE recipes SET title = :title, ingredients = :ingredients, cook_time_min = :cook_time_min WHERE id = :id');
$stmt->execute([
    ':id'            => $id,
    ':title'         => $title,
    ':ingredients'   => $ingredients,
    ':cook_time_min' => $cookTimeMin,
]);
header('Location: index.php?msg=updated');
exit;
```

### Крок 6. Видалення рецепта (DELETE)
У файлі `delete.php` виконується запит:
```php
$stmt = $pdo->prepare('DELETE FROM recipes WHERE id = :id');
$stmt->execute([':id' => $id]);
header('Location: index.php?msg=deleted');
exit;
```
Перед видаленням користувач підтверджує дію через JavaScript `confirm()`.

### Крок 7. Захист від SQL-ін'єкцій
Завдяки розділенню коду запиту та параметрів через `prepare()` та `execute()` введення рядків кшталту:
```

O'Connor's Cake'); DROP TABLE recipes; --
```
не змінює синтаксичного дерева SQL-запиту, а зберігається та шукається виключно як звичайний текст.

    exit;
}

## 🛠 Реалізований функціонал

1.  **Живий пошук (Live Search):**
    При введенні символів у поле пошуку JavaScript відправляє асинхронний `GET` запит до `api_list.php?q=...`. Сервер виконує SQL `LIKE` запит та повертає JSON. Сторінка оновлюється миттєво.
2.  **Додавання без перезавантаження:**
    Форма додавання перехоплюється через `event.preventDefault()`. Дані збираються в об'єкт, серіалізуються в JSON та відправляються `POST` запитом до `api_add.php`. Після успіху список рецептів оновлюється автоматично.
3.  **Видалення через AJAX:**
    Кнопки видалення мають атрибут `data-id`. При натисканні відправляється запит до `api_delete.php`. Рядок таблиці зникає без перезавантаження всієї сторінки.
4.  **REST API Backend:**
    Використано `header('Content-Type: application/json')` для всіх API файлів. Обробка JSON-вводу через `file_get_contents('php://input')`.

document.addEventListener('DOMContentLoaded', () => {

    const resultsContainer = document.getElementById('results');

    // Завантажити всі рецепти

    });

    // Додавання через AJAX
    addForm.addEventListener('submit', async (e) => {
        e.preventDefault();
        const data = {
            title: document.getElementById('title').value,
            ingredients: document.getElementById('ingredients').value,
            cook_time_min: parseInt(document.getElementById('cook_time_min').value)
        };

        try {
            const res = await fetch('api_add.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(data)
            });

            if (!res.ok) {
                const err = await res.json();
                throw new Error(err.error || 'Помилка');
            }

            addForm.reset();
            loadRecipes(searchInput.value.trim()); // Оновлюємо список
        } catch (err) {
            alert('Помилка: ' + err.message);
        }
    });

    // Видалення рецепта через AJAX (делегування подій)
    resultsContainer.addEventListener('click', async (e) => {
        if (e.target.classList.contains('btn-delete')) {
            const id = e.target.dataset.id;
            if (!confirm('Ви дійсно хочете видалити цей рецепт?')) return;

            try {
                const res = await fetch('api_delete.php', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ id: parseInt(id) })
                });

                if (!res.ok) {
                    const err = await res.json();
                    throw new Error(err.error || 'Помилка видалення');
                }

                loadRecipes(searchInput.value.trim());
            } catch (err) {
                alert('Помилка: ' + err.message);
            }
        }
    });

    async function loadRecipes(query = '') {
        try {
            const url = query ? `api_list.php?q=${encodeURIComponent(query)}` : 'api_list.php';
            const res = await fetch(url);
            const recipes = await res.json();
            renderRecipes(recipes);
        } catch (err) {
            resultsContainer.innerHTML = `<p style="color:red">Помилка завантаження: ${err.message}. Перевірте чи налаштована БД через <a href="setup.php">setup.php</a>.</p>`;
        }
    }

    function renderRecipes(recipes) {
        if (!recipes.length) {
            resultsContainer.innerHTML = '<p class="empty-msg">Рецептів не знайдено.</p>';
            return;
        }

        let html = '<table class="recipes-table"><thead><tr><th>ID</th><th>Назва</th><th>Інгредієнти</th><th>Час (хв)</th><th>Дії</th></tr></thead><tbody>';
        recipes.forEach(r => {
            html += `<tr>
                <td>${r.id}</td>
                <td><strong>${escapeHtml(r.title)}</strong></td>
                <td>${escapeHtml(r.ingredients)}</td>
                <td><span class="badge ${r.cook_time_min <= 20 ? 'badge-quick' : ''}">${r.cook_time_min} хв</span></td>
                <td><button class="btn-danger btn-delete" data-id="${r.id}">🗑 Видалити</button></td>
            </tr>`;
        });
        html += '</tbody></table>';
        resultsContainer.innerHTML = html;
    }

    function escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;

# API Documentation — Практична робота №6

**Варіант №6:** Кулінарна книга (рецепти, інгредієнти)
**Єдина точка входу:** `api.php`
**Формат відповідей:** `application/json`

---

## Загальна структура відповіді

### Успішна відповідь:
```json
{
  "success": true,
  "data": { ... }
}
```

### Помилка:
```json
{
  "success": false,
  "error": "Опис помилки"
}
```

---

## Перелік підтримуваних ендпоінтів

| HTTP Метод | Параметри URL | Тіло запиту (JSON) | Код статусу | Опис |
|------------|---------------|-------------------|-------------|------|
| `GET` | `resource=recipes` | — | `200 OK` | Отримати список усіх рецептів |
| `GET` | `resource=recipes&id={id}` | — | `200 OK` / `404 Not Found` | Отримати один рецепт за ID |
| `GET` | `resource=recipes&action=shortest` | — | `200 OK` / `404 Not Found` | **Доменна дія:** найшвидший рецепт |
| `GET` | `resource=recipes&action=search&q={text}` | — | `200 OK` | **Доменна дія:** пошук рецептів за інгредієнтом |
| `POST` | `resource=recipes` | `{"title": "...", "ingredients": "...", "cook_time_min": 20}` | `201 Created` / `400 Bad Request` | Створити новий рецепт |
| `POST` | `resource=recipes&action=convert_units` | `{"ingredients_text": "...", "target": "oz"}` | `200 OK` / `400 Bad Request` | **Доменна дія:** перерахунок мір інгредієнтів (грами ↔ унції) |

---

## Приклади тестування через `curl`

### 1. Отримати всі рецепти (GET)
```bash
curl -i -X GET "http://localhost:8000/api.php?resource=recipes"
```

### 2. Отримати один рецепт за ID (GET)
```bash
curl -i -X GET "http://localhost:8000/api.php?resource=recipes&id=1"
```

### 3. Знайти найшвидший рецепт (Доменна дія GET)
```bash
curl -i -X GET "http://localhost:8000/api.php?resource=recipes&action=shortest"
```

### 4. Пошук за інгредієнтом (Доменна дія GET)
```bash
curl -i -X GET "http://localhost:8000/api.php?resource=recipes&action=search&q=буряк"
```

### 5. Створити новий рецепт (POST)
```bash
curl -i -X POST "http://localhost:8000/api.php?resource=recipes" \
  -H "Content-Type: application/json" \
  -d '{"title": "Омлет з сиром", "ingredients": "Яйця, сир, масло, сіль", "cook_time_min": 10}'
```

### 6. Перерахувати міри інгредієнтів в унції (Доменна дія POST)
```bash
curl -i -X POST "http://localhost:8000/api.php?resource=recipes&action=convert_units" \
  -H "Content-Type: application/json" \
  -d '{"ingredients_text": "Борошно 200г, цукор 100г", "target": "oz"}'
```

/**
 * Практична робота №6: Розробка власного API
 * Варіант №6: Кулінарна книга
 *
 * Крок 1-2. Єдина точка входу, заголовки, маршрутизація.

// Функція для стандартизованої відповіді
function sendResponse(bool $success, $data_or_error = null, int $status = 200) {
    http_response_code($status);
    $response = ['success' => $success];
    if ($success) {
        $response['data'] = $data_or_error;
    } else {
        $response['error'] = $data_or_error;
    }
    echo json_encode($response, JSON_UNESCAPED_UNICODE);
    exit;
}

// Перевірка ресурсу
if ($resource !== 'recipes') {
    sendResponse(false, 'Невідомий ресурс. Доступні ресурси: recipes', 404);
}

// МАРШРУТИЗАЦІЯ (Крок 3-7)
switch ($method) {
    case 'GET':
        handleGet($pdo, $id, $action);
        break;
    case 'POST':
        handlePost($pdo, $action);
        break;
    default:
        sendResponse(false, "Метод $method не підтримується для цього ресурсу", 405);
        break;
}

/**
 * Крок 4. Реалізація GET (Список, Одиничний запис, Пошук)
 */
function handleGet(PDO $pdo, ?int $id, ?string $action) {
    // 1. Отримати одиничний запис за ID
    if ($id !== null) {
        $stmt = $pdo->prepare("SELECT * FROM recipes WHERE id = ?");
        $stmt->execute([$id]);
        $recipe = $stmt->fetch();
        if ($recipe) {
            sendResponse(true, $recipe);
        } else {
            sendResponse(false, "Рецепт з ID $id не знайдено", 404);
        }
    }

    // 2. Список усіх рецептів
    $stmt = $pdo->query("SELECT * FROM recipes ORDER BY id DESC");
    $recipes = $stmt->fetchAll();
    sendResponse(true, $recipes);
}

/**
 * Крок 5-6. Реалізація POST (Створення, Доменна дія search)
 */
function handlePost(PDO $pdo, ?string $action) {
    // Отримуємо дані з тіла запиту (JSON або Form Data)
    $input = json_decode(file_get_contents('php://input'), true) ?? $_POST;

    // Крок 6. Доменна дія: action=search (пошук за часом приготування)
    if ($action === 'search') {
        $max_time = $input['max_time'] ?? $_GET['max_time'] ?? null;
        if ($max_time === null || !is_numeric($max_time)) {
            sendResponse(false, "Параметр 'max_time' є обов'язковим для пошуку та має бути числом", 400);
        }

        $stmt = $pdo->prepare("SELECT * FROM recipes WHERE cook_time_min <= ? ORDER BY cook_time_min ASC");
        $stmt->execute([(int)$max_time]);
        $found = $stmt->fetchAll();
        sendResponse(true, $found);
    }

    // Крок 5. Створення нового запису (якщо немає action)
    if ($action === null) {
        $title = trim((string)($input['title'] ?? ''));
        $ingredients = trim((string)($input['ingredients'] ?? ''));
        $cook_time_min = $input['cook_time_min'] ?? null;

        // Перевірка обов'язкових полів (Крок 7)
        if ($title === '' || $ingredients === '' || $cook_time_min === null) {
            sendResponse(false, "Відсутні обов'язкові поля: title, ingredients, cook_time_min", 400);
        }

        if (!is_numeric($cook_time_min) || (int)$cook_time_min <= 0) {
            sendResponse(false, "Поле 'cook_time_min' має бути числом більше 0", 400);
        }

        try {
            $stmt = $pdo->prepare("INSERT INTO recipes (title, ingredients, cook_time_min) VALUES (?, ?, ?)");
            $stmt->execute([$title, $ingredients, (int)$cook_time_min]);
            $newId = $pdo->lastInsertId();

            sendResponse(true, [
                'id' => $newId,
                'title' => $title,
                'ingredients' => $ingredients,
                'cook_time_min' => (int)$cook_time_min
            ], 201);
        } catch (PDOException $e) {
            sendResponse(false, "Помилка бази даних: " . $e->getMessage(), 500);
        }
    }

    sendResponse(false, "Невідома дія (action). Доступні дії для POST: search або порожня дія для створення.", 400);
}
